" 行号
set number

" 关闭自动注释（对粘贴多行有用）
set paste

" 设置颜色主题
colorscheme desert

set backspace=2

" 语法高亮
syntax on

" 智能对齐
set smartindent

" 在处理未保存或只读文件时，弹出确认
set confirm

" tab 宽度
set tabstop=4

" 统一缩进（为4）
set softtabstop=4
set shiftwidth=4

" 历史记录数
set history=50

" 搜索逐字符高亮
set hlsearch
set incsearch

" 编码设置
set encoding=utf-8

" 设置语言
set langmenu=zn_CN.UTF-8
set helplang=cn

" 设置命令行的高度
set cmdheight=1

" 在编辑时，右下角显示光标位置的状态行
set ruler
" 光标在移动到 buffer 的顶部与底部时保持的距离
set scrolloff=3
 
" 高亮显示对应的括号
set showmatch

" 对应括号高亮的时间（单位是十分之一秒）
set matchtime=5

" 总是显示状态行
set laststatus=2

" 高亮显示多余的空格或制表符
highlight ExtraWhitespace ctermbg=red guibg=red
autocmd BufWinEnter * match ExtraWhitespace /\s+$\| \+\ze\t\+\|\t\+\zs \+/

修改 ~/.vimrc 以使用，或修改 /etc/vim/vimrc（Ubuntu 20.04）来为所有用户应用。
